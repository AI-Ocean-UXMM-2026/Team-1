================================================================================
QUICK START COMMANDS
================================================================================

# Install Remotion
npm install remotion @remotion/cli

# Start Interactive Studio
npm run dev
# Opens: http://localhost:3000

# Render a video
npx remotion render Demo1_BasicAnimation

# Render all compositions
npx remotion render

# Render with options
npx remotion render --quality=90 --codec=h264 --concurrency=4

# Export to PNG sequence
npx remotion render --sequence

# Export to WebM
npx remotion render --codec=vp9
